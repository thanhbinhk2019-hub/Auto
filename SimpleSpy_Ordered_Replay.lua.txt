local function runStep(logNo, name, args, callFn, isInvoke)
    if stopped then return false end
    label.Text = string.format("Đang chạy Log #%d/157: %s", logNo, name)
    local ok, result = pcall(callFn, args)
    if not ok then
        label.Text = string.format("LỖI tại Log #%d (%s) — đã dừng", logNo, name)
        warn("[Replay] Dừng tại Log #" .. logNo .. ": " .. tostring(result))
        stopped = true
        return false
    end
    if isInvoke then
        print(string.format("[Replay] Log #%d %s: InvokeServer trả về %s", logNo, name, tostring(result)))
        label.Text = string.format("Log #%d: %s — đã phản hồi", logNo, name)
    else
        print(string.format("[Replay] Log #%d %s: đã gửi (FireServer không có xác nhận)", logNo, name))
        label.Text = string.format("Log #%d: %s — đã gửi", logNo, name)
    end
    task.wait(delayBetween)
    return not stopped
end
